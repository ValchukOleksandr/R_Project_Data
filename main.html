calculate_outliers <- function(x) {
  n <- sum(!is.na(x))
  
  q1 <- quantile(x, 0.25, na.rm = TRUE)
  q3 <- quantile(x, 0.75, na.rm = TRUE)
  iqr <- q3 - q1
  
  lower_iqr <- q1 - 1.5 * iqr
  upper_iqr <- q3 + 1.5 * iqr
  
  avg <- mean(x, na.rm = TRUE)
  st_dev <- sd(x, na.rm = TRUE)
  
  lower_z <- avg - 3 * st_dev
  upper_z <- avg + 3 * st_dev
  
  outliers_iqr <- sum(x < lower_iqr | x > upper_iqr, na.rm = TRUE)
  outliers_z <- sum(x < lower_z | x > upper_z, na.rm = TRUE)
  
  data.frame(
    n = n,
    missing = sum(is.na(x)),
    lower_iqr = round(lower_iqr, 2),
    upper_iqr = round(upper_iqr, 2),
    outliers_iqr = outliers_iqr,
    pct_iqr = round(outliers_iqr / n * 100, 2),
    lower_z = round(lower_z, 2),
    upper_z = round(upper_z, 2),
    outliers_z = outliers_z,
    pct_z = round(outliers_z / n * 100, 2),
    row.names = NULL
  )
}
